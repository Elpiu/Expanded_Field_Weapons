# Expanded Field Weapons - on_actions
#
# OTTIMIZZAZIONE CPU
# Prima la mod aveva tre blocchi on_daily: ogni giorno, per ogni paese,
# circa 10 controlli di tecnologie e flag, per tutta la partita.
# Ora:
#   - l'inizializzazione del giocatore (aereo in legno e navi improvvisate)
#     è una catena di eventi nascosti lanciata una volta sola
#     (events/efw_welcome_events.txt);
#   - la sincronizzazione delle tecnologie gira una volta a settimana e,
#     quando un paese ha sbloccato tutto, si riduce a un solo controllo
#     di flag (efw_sync_complete_v2; il suffisso si incrementa quando si aggiungono sblocchi).
# Risultato: circa 7 volte meno valutazioni all'inizio e quasi zero a fine
# progressione. Costo: uno sblocco può arrivare fino a 7 giorni dopo la ricerca.

on_actions = {

    # =========================================================
    # STARTUP (non gira su un paese specifico)
    # =========================================================

    on_startup = {
        effect = {

            # Pipe Gun e mezzi leggeri improvvisati disponibili a tutti dal primo giorno.
            # enable_equipments/enable_subunits esistono solo nelle tecnologie,
            # quindi si assegnano tecnologie nascoste.
            every_country = {
                efw_unlock_improvised_gear = yes
            }

            # Giocatori umani: inizializzazione + evento di benvenuto.
            every_country = {
                limit = {
                    is_ai = no
                }
                efw_player_init = yes
            }
        }
    }


    # =========================================================
    # SINCRONIZZAZIONE SETTIMANALE (gira su ogni paese)
    # =========================================================

    on_weekly = {
        effect = {

            # Cambio di nazione o nuovo giocatore in multiplayer:
            # inizializza anche un paese preso a partita in corso.
            if = {
                limit = {
                    is_ai = no
                    NOT = { has_country_flag = efw_player_init_started }
                }
                efw_player_init = yes
            }

            # Paesi che hanno già sbloccato tutto: un solo controllo e basta.
            if = {
                limit = {
                    NOT = { has_country_flag = efw_sync_complete_v2 }
                }
                efw_sync_support_techs = yes
            }
        }
    }
}
