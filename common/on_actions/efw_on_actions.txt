# Expanded Field Weapons v1.2.0
# Daily synchronization for the core EFW infantry/support equipment.
#
# This file intentionally applies the hidden progression technologies
# to both human and AI countries.
#
# Aircraft and naval improvised variants are handled separately
# in their dedicated on_action files.

on_actions = {

    # =========================================================
    # STARTUP
    # =========================================================

    on_startup = {
        effect = {

            # Sblocca la Pipe Gun per tutte le nazioni del gioco
            every_country = {
                enable_equipments = { efw_pipe_gun_rifle }
            }

            # Show the EFW startup confirmation only to human players.
            every_country = {
                limit = {
                    is_ai = no
                }

                country_event = {
                    id = efw_test.1
                    days = 1
                }
            }
        }
    }


    # =========================================================
    # DAILY TECHNOLOGY SYNCHRONIZATION
    # =========================================================

    on_daily = {
        effect = {

            # -------------------------------------------------
            # SUPPORT WEAPONS I
            # Unlock the permanent EFW equipment and formations.
            # -------------------------------------------------

            if = {
                limit = {
                    has_tech = support_weapons

                    NOT = {
                        has_tech = efw_unlock_core_weapons
                    }
                }

                set_technology = {
                    efw_unlock_core_weapons = 1
                    popup = no
                }
            }


            # -------------------------------------------------
            # SUPPORT WEAPONS II
            # -------------------------------------------------

            if = {
                limit = {
                    has_tech = support_weapons2

                    NOT = {
                        has_tech = efw_support_upgrade_2
                    }
                }

                set_technology = {
                    efw_support_upgrade_2 = 1
                    popup = no
                }
            }


            # -------------------------------------------------
            # SUPPORT WEAPONS III
            # -------------------------------------------------

            if = {
                limit = {
                    has_tech = support_weapons3

                    NOT = {
                        has_tech = efw_support_upgrade_3
                    }
                }

                set_technology = {
                    efw_support_upgrade_3 = 1
                    popup = no
                }
            }


            # -------------------------------------------------
            # SUPPORT WEAPONS IV
            # -------------------------------------------------

            if = {
                limit = {
                    has_tech = support_weapons4

                    NOT = {
                        has_tech = efw_support_upgrade_4
                    }
                }

                set_technology = {
                    efw_support_upgrade_4 = 1
                    popup = no
                }
            }


            # -------------------------------------------------
            # VANILLA INFANTRY AT I
            # -------------------------------------------------

            if = {
                limit = {
                    has_tech = infantry_at

                    NOT = {
                        has_tech = efw_at_upgrade_1
                    }
                }

                set_technology = {
                    efw_at_upgrade_1 = 1
                    popup = no
                }
            }


            # -------------------------------------------------
            # VANILLA INFANTRY AT II
            # -------------------------------------------------

            if = {
                limit = {
                    has_tech = infantry_at2

                    NOT = {
                        has_tech = efw_at_upgrade_2
                    }
                }

                set_technology = {
                    efw_at_upgrade_2 = 1
                    popup = no
                }
            }
        }
    }
}
