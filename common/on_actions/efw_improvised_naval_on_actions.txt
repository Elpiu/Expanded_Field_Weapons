# EFW Naval Improvised v0.4 EASY
#
# This deliberately uses VANILLA hull types:
#   ship_hull_light_1
#   ship_hull_submarine_1
#
# create_equipment_variant with those hulls is known to work in the user's game.

on_actions = {

    on_daily = {
        effect = {

            # Unlock our primitive modules.
            if = {
                limit = {
                    NOT = {
                        has_tech = efw_improvised_naval_modules_unlock
                    }
                }

                set_technology = {
                    efw_improvised_naval_modules_unlock = 1
                    popup = no
                }
            }


            # Create the two variants exactly once.
            if = {
                limit = {
                    has_tech = efw_improvised_naval_modules_unlock

                    NOT = {
                        has_country_flag = efw_improvised_naval_v04_created
                    }
                }

                # =====================================================
                # IMPROVISED MOTORBOAT
                # Vanilla destroyer hull + EFW low-cost modules.
                # =====================================================
                create_equipment_variant = {
                    name = "EFW Improvised Motorboat"

                    type = ship_hull_light_1

                    parent_version = 0
                    show_position = no
                    obsolete = no
                    allow_without_tech = yes
                    role_icon_index = auto

                    icon = "GFX_efw_motorboat_medium"

                    modules = {
                        fixed_ship_battery_slot = efw_motorboat_machinegun
                        fixed_ship_anti_air_slot = empty
                        fixed_ship_fire_control_system_slot = empty
                        fixed_ship_radar_slot = empty
                        fixed_ship_engine_slot = efw_motorboat_engine
                        fixed_ship_torpedo_slot = empty
                        mid_1_custom_slot = empty
                        rear_1_custom_slot = empty
                    }
                }


                # =====================================================
                # ONE-MAN MIDGET SUBMARINE
                # Vanilla submarine hull + tiny EFW engine and torpedoes.
                # Two tiny torpedo mounts give it a real offensive niche.
                # =====================================================
                create_equipment_variant = {
                    name = "EFW One-Man Midget Submarine"

                    type = ship_hull_submarine_1

                    parent_version = 0
                    show_position = no
                    obsolete = no
                    allow_without_tech = yes
                    role_icon_index = auto

                    icon = "GFX_efw_midget_sub_medium"

                    modules = {
                        fixed_ship_torpedo_slot = efw_midget_sub_torpedo
                        fixed_ship_engine_slot = efw_midget_sub_engine
                        rear_1_custom_slot = efw_midget_sub_torpedo
                    }
                }


                set_country_flag = efw_improvised_naval_v04_created
            }
        }
    }
}
